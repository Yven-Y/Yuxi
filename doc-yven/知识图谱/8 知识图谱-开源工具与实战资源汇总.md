## 二、Python 实战代码示例

### 2.1 使用 Neo4j 创建知识图谱

```python
from neo4j import GraphDatabase

class KnowledgeGraph:
    def __init__(self, uri, user, password):
        self.driver = GraphDatabase.driver(uri, auth=(user, password))

    def add_entity(self, name, entity_type, properties=None):
        props = properties or {}
        query = f"""
        MERGE (e:{entity_type} {{name: $name}})
        SET e += $props
        RETURN e
        """
        with self.driver.session() as session:
            return session.run(query, name=name, props=props).single()

    def add_relation(self, from_name, to_name, relation_type, props=None):
        props = props or {}
        query = f"""
        MATCH (a {{name: $from_name}}), (b {{name: $to_name}})
        MERGE (a)-[r:{relation_type}]->(b)
        SET r += $props
        RETURN r
        """
        with self.driver.session() as session:
            return session.run(query, from_name=from_name,
                             to_name=to_name, props=props).single()

    def query(self, cypher):
        with self.driver.session() as session:
            return list(session.run(cypher))

# 使用示例
kg = KnowledgeGraph("bolt://localhost:7687", "neo4j", "password")

# 添加实体
kg.add_entity("张三", "Person", {"age": 30})
kg.add_entity("阿里巴巴", "Company", {"location": "杭州"})

# 添加关系
kg.add_relation("张三", "阿里巴巴", "WORKS_AT", {"since": 2015})

# 查询
result = kg.query("""
    MATCH (p:Person)-[:WORKS_AT]->(c:Company)
    RETURN p.name, c.name
""")
```

### 2.2 使用 LightRAG 快速构建 GraphRAG

```python
from lightrag import LightRAG, QueryParam

# 初始化
rag = LightRAG(working_dir="./kg_storage")

# 插入文档
rag.insert("""
阿里巴巴由马云于1999年在杭州创立。
马云毕业于杭州师范大学英语专业。
阿里巴巴旗下有淘宝、天猫、阿里云等业务。
""")

# 查询
result = rag.query(
    "阿里巴巴的创始人毕业于哪所学校？",
    param=QueryParam(mode="hybrid")
)
print(result)
```

### 2.3 使用 LangChain + Neo4j 构建 GraphRAG

```python
from langchain_community.graphs import Neo4jGraph
from langchain_openai import ChatOpenAI
from langchain.chains import GraphCypherQAChain

# 连接图数据库
graph = Neo4jGraph(
    url="bolt://localhost:7687",
    username="neo4j",
    password="password"
)

# 初始化 LLM
llm = ChatOpenAI(model="gpt-4")

# 创建 QA Chain
chain = GraphCypherQAChain.from_llm(
    llm=llm,
    graph=graph,
    verbose=True
)

# 问答
result = chain.invoke({"query": "阿里巴巴的创始人是谁？"})
print(result)
```

### 2.4 使用 LLM 自动抽取知识

```python
from openai import OpenAI
import json

client = OpenAI()

def extract_knowledge(text):
    prompt = f"""
    从以下文本中提取实体和关系，返回JSON格式：

    文本：{text}

    要求：
    1. 实体格式：[{{"name": "名称", "type": "类型"}}]
    2. 关系格式：[{{"head": "头实体", "relation": "关系", "tail": "尾实体"}}]
    3. 实体类型可选：Person, Company, Location, Product, Event
    4. 关系类型用中文或英文动词

    只返回JSON，不要其他内容。
    """

    response = client.chat.completions.create(
        model="gpt-4",
        messages=[{"role": "user", "content": prompt}]
    )

    content = response.choices[0].message.content
    return json.loads(content)

# 使用
text = "马云创立了阿里巴巴，公司总部位于杭州。"
knowledge = extract_knowledge(text)
print(json.dumps(knowledge, ensure_ascii=False, indent=2))
```
